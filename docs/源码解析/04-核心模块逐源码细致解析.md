### 4.2 RAG向量知识库模块

Vanna 2.0 的 RAG 模块不再是传统的 "训练 schema → 向量化 DDL → 检索 schema"，而是基于 **AgentMemory** 的能力接口，存储和检索 **工具使用模式**（ToolMemory）和 **文本知识**（TextMemory）。

**两类记忆对比：**

| 维度 | ToolMemory | TextMemory |
|------|-----------|------------|
| 用途 | 记录成功的工具调用模式（SQL+参数） | 记录自由文本知识（表结构说明、业务规则等） |
| 检索方式 | 按 question 向量相似度 + tool_name 过滤 | 按 content 向量相似度 |
| 存储字段 | question, tool_name, args_json, success, timestamp | content, timestamp, is_text_memory=True |
| 使用场景 | Few-shot 召回（LLM 调用 run_sql 前参考） | 自动增强 System Prompt |
| 触发时机 | LLM 通过 save_question_tool_args 工具调用 | LLM 通过 save_text_memory 工具调用 / 代码直接调用 |

```python
def _get_collection(self):
    if self._collection is None:
        client = self._get_client()  # 懒加载 PersistentClient
        try:
            # 优先获取已有 Collection（不重新下载 embedding 模型）
```

**save_tool_usage 核心逻辑：**

```python
# 第166-199行
async def save_tool_usage(self, question, tool_name, args, context, success=True, metadata=None):
    def _save():
        collection = self._get_collection()
        memory_data = {
            "question": question,             # 用作向量化文档
            "tool_name": tool_name,
            "args_json": json.dumps(args),    # 复杂对象序列化为 JSON 字符串
            "timestamp": datetime.now().isoformat(),
            "success": success,
            "metadata_json": json.dumps(metadata or {}),
        }
        # upsert: question → embedding → 存入 ChromaDB
        collection.upsert(ids=[memory_id], documents=[question], metadatas=[memory_data])
    # 线程池异步执行（ChromaDB 不支持 async）
    await asyncio.get_event_loop().run_in_executor(self._executor, _save)
```

**search_similar_usage 核心逻辑：**

```python
# 第201-265行
async def search_similar_usage(self, question, context, limit=10,
                               similarity_threshold=0.7, tool_name_filter=None):
    def _search():
        collection = self._get_collection()
        # 过滤条件：仅返回成功的、指定工具的记录
        if tool_name_filter:
            where_filter = {"$and": [{"success": True}, {"tool_name": tool_name_filter}]}
        else:
            where_filter = {"success": True}

        # 向量检索：question → embedding → 相似度搜索
        results = collection.query(query_texts=[question], n_results=limit, where=where_filter)

        for i, (id_, distance, metadata) in enumerate(zip(...)):
            # ChromaDB 返回 L2 距离，转换为相似度分数
            similarity_score = max(0, 1 - distance)
            if similarity_score >= similarity_threshold:
                args = json.loads(metadata.get("args_json", "{}"))
                memory = ToolMemory(memory_id=id_, question=metadata["question"],
                                    tool_name=metadata["tool_name"], args=args, ...)
                search_results.append(ToolMemorySearchResult(memory=memory, ...))
        return search_results
    return await asyncio.get_event_loop().run_in_executor(self._executor, _search)
```

**向量化原理：**
- ChromaDB 默认使用 `DefaultEmbeddingFunction()`（基于 ONNX 模型 all-MiniLM-L6-v2）
- 将 `question` 文本转为 384 维向量
- 使用 L2 距离度量相似度，转换为 `similarity_score = 1 - distance`
- 默认阈值 `similarity_threshold = 0.7`

```python
class DefaultLlmContextEnhancer(LlmContextEnhancer):
    def __init__(self, agent_memory: Optional[AgentMemory] = None):
        self.agent_memory = agent_memory

    async def enhance_system_prompt(self, system_prompt: str, user_message: str, user: User) -> str:
        if not self.agent_memory:
            return system_prompt  # 无 AgentMemory，不做增强

        context = ToolContext(user=user, conversation_id="temp",
                              request_id=str(uuid.uuid4()), agent_memory=self.agent_memory)
        # 从向量库检索与用户问题相关的文本记忆
        memories = await self.agent_memory.search_text_memories(
            query=user_message, context=context, limit=5
        )
        if not memories:
            return system_prompt

        examples_section = "\n\n## Relevant Context from Memory\n\n"
        for result in memories:
            examples_section += f"• {result.memory.content}\n"
        return system_prompt + examples_section
```

```python
class LlmService(ABC):
    """LLM 服务抽象接口，所有 LLM 厂商实现必须继承"""

    @abstractmethod
    async def send_request(self, request: LlmRequest) -> LlmResponse:
        """发送非流式请求，返回完整响应"""
        ...

    async def stream_request(self, request: LlmRequest) -> AsyncGenerator[LlmStreamChunk, None]:
        """发送流式请求，逐步返回响应块（默认实现：调用 send_request 模拟）"""
        response = await self.send_request(request)
        if response.content:
            yield LlmStreamChunk(content=response.content)
        if response.tool_calls:
            yield LlmStreamChunk(tool_calls=response.tool_calls, finish_reason=response.finish_reason)

    async def validate_tools(self, tools: List[ToolSchema]) -> List[str]:
        """校验工具 Schema 合法性，返回错误列表"""
        return []
```

```python
# 第190-242行
def _build_payload(self, request: LlmRequest) -> Dict[str, Any]:
    messages = []
    # 1. System Prompt 作为第一条消息
    if request.system_prompt:
        messages.append({"role": "system", "content": request.system_prompt})

    # 2. 转换对话历史
    for m in request.messages:
        msg = {"role": m.role, "content": m.content}
        # tool 消息需要 tool_call_id
        if m.role == "tool" and m.tool_call_id:
            msg["tool_call_id"] = m.tool_call_id
        # assistant 消息需要 tool_calls（OpenAI 格式）
        elif m.role == "assistant" and m.tool_calls:
            msg["tool_calls"] = [
                {"id": tc.id, "type": "function",
                 "function": {"name": tc.name, "arguments": json.dumps(tc.arguments)}}
                for tc in m.tool_calls
            ]
        messages.append(msg)

    # 3. 转换工具 Schema 为 OpenAI Function 格式
    tools_payload = [{
        "type": "function",
        "function": {"name": t.name, "description": t.description, "parameters": t.parameters}
    } for t in request.tools] if request.tools else None

    return {"model": self.model, "messages": messages, "tools": tools_payload, ...}
```

```python
async def stream_request(self, request: LlmRequest) -> AsyncGenerator[LlmStreamChunk, None]:
    stream = self._client.chat.completions.create(**payload, stream=True)
    tc_builders: Dict[int, Dict] = {}  # 按索引累积 tool_calls 增量

    for event in stream:
        delta = event.choices[0].delta
        if delta.content:
            yield LlmStreamChunk(content=delta.content)  # 逐步 yield 文本增量

        # 累积流式 tool_calls
        for tc in getattr(delta, "tool_calls", None) or []:
            b = tc_builders.setdefault(tc.index, {"id": None, "name": None, "arguments": ""})
            if tc.id: b["id"] = tc.id
            if tc.function and tc.function.name: b["name"] = tc.function.name
            if tc.function and tc.function.arguments: b["arguments"] += tc.function.arguments

    # 流结束后，emit 最终的 tool_calls
    if final_tool_calls:
```

源码位置：[tools/run_sql.py](file:///d:/workspace/sourceWorkspace/vanna/src/vanna/tools/run_sql.py)

```python
class RunSqlTool(Tool[RunSqlToolArgs]):
    def __init__(self, sql_runner: SqlRunner, file_system: Optional[FileSystem] = None):
        self.sql_runner = sql_runner
        self.file_system = file_system

    @property
    def name(self) -> str:
        return "run_sql"

    @property
    def description(self) -> str:
        return "Execute a SQL query against the database"

    async def execute(self, context, args: RunSqlToolArgs) -> ToolResult:
        try:
            df = await self.sql_runner.run_sql(args, context)
            query_type = args.sql.strip().upper().split()[0]

            if query_type == "SELECT":
                # SELECT: 返回 DataFrame + CSV 文本
                results_data = df.to_dict("records")
                csv_content = df.to_csv(index=False)
                # 截断超长结果（>1000 字符）
                results_preview = csv_content[:1000] + "\n(Results truncated...)" if len(csv_content) > 1000 else csv_content

                return ToolResult(
                    success=True,
                    result_for_llm=results_preview,  # 给 LLM 的文本
                    ui_component=UiComponent(rich_component=DataFrameComponent.from_records(...)),
                    metadata={"row_count": len(df), "columns": df.columns.tolist()},
                )
            else:
                # INSERT/UPDATE/DELETE: 返回影响行数
                rows_affected = len(df) if not df.empty else 0
                return ToolResult(success=True, result_for_llm=f"Query executed. {rows_affected} row(s) affected.")

        except Exception as e:
            return ToolResult(success=False, result_for_llm=f"Error executing query: {str(e)}", error=str(e))
```

#### 4.5.3 PostgreSQL 实现

源码位置：[integrations/postgres/sql_runner.py](file:///d:/workspace/sourceWorkspace/vanna/src/vanna/integrations/postgres/sql_runner.py)

```python
class PostgresRunner(SqlRunner):
    def __init__(self, connection_string=None, **connection_params):
        self.connection_string = connection_string
        self.connection_params = connection_params

    async def run_sql(self, args: RunSqlToolArgs, context: ToolContext) -> pd.DataFrame:
        if self.connection_string:
            conn = psycopg2.connect(self.connection_string)
        else:
            conn = psycopg2.connect(**self.connection_params)

        cursor = conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor)
        try:
            cursor.execute(args.sql)

            query_type = args.sql.strip().upper().split()[0]
            if query_type == "SELECT":
                rows = cursor.fetchall()
                return pd.DataFrame([dict(row) for row in rows])
            else:
                conn.commit()
                return pd.DataFrame({"rows_affected": [cursor.rowcount]})
        finally:
            cursor.close()
            conn.close()
```

### 4.6 错误重试与自优化模块

Vanna 2.0 的错误处理采用 **Agent 工具循环 + LLM 自我纠错** 机制，辅以可插拔的 `LlmMiddleware` 和 `ErrorRecoveryStrategy`。

#### 4.6.1 Agent 工具循环中的错误恢复

核心流程（[core/agent/agent.py](file:///d:/workspace/sourceWorkspace/vanna/src/vanna/core/agent/agent.py)）：

```
while tool_iterations < max_tool_iterations:
    response = await llm_service.send_request(request)

    if response.is_tool_call():
        tool_iterations += 1
        for tool_call in response.tool_calls:
            result = await tool_registry.execute(tool_call, context)

            # 关键：失败结果也会添加为 tool 消息
            conversation.add_message(Message(
                role="tool",
                content=result.result_for_llm,  # 错误信息
                tool_call_id=tool_call.id
            ))

        # 重建请求（含错误信息），LLM 看到错误后自动修正
        request = await self._build_llm_request(...)
    else:
        break  # 最终回复
```

**为什么 LLM 能自动纠错？**
- 失败的工具结果以 `Message(role="tool", content="Error: column 'regon' does not exist")` 形式追加到对话历史
- LLM 看到自己生成的 SQL 导致错误 + 错误信息 → 自然推断出需要修正 → 生成修正后的 SQL → 再次调用 run_sql

```python
class ErrorRecoveryStrategy(ABC):
    async def handle_tool_error(self, error: Exception, context: ToolContext, attempt: int) -> RecoveryAction:
        """处理工具执行错误"""
        return RecoveryAction(action=RecoveryActionType.FAIL, message=f"Tool error: {str(error)}")

    async def handle_llm_error(self, error: Exception, request: LlmRequest, attempt: int) -> RecoveryAction:
        """处理 LLM 调用错误"""
```

```python
class ExponentialBackoffStrategy(ErrorRecoveryStrategy):
```
