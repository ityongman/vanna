# Vanna AI 源码深度解析文档

## 2. 整体分层架构设计（自上而下）

Vanna 2.0 采用 **七层分层架构**，自上而下依次为：用户接入层 → 会话管理层 → RAG知识库检索层 → SQL生成内核层 → SQL校验安全层 → 数据库执行层 → 结果解析复盘层。每层职责清晰，通过接口抽象实现层间解耦。

```python
async def enhance_system_prompt(self, system_prompt, user_message, user):
    # 创建临时 ToolContext 用于内存搜索
    context = ToolContext(
        user=user, conversation_id="temp", request_id=str(uuid.uuid4()),
        agent_memory=self.agent_memory,
    )
    # 核心：从向量库中搜索与用户问题相关的文本记忆
    memories = await self.agent_memory.search_text_memories(
        query=user_message, context=context, limit=5
    )
    if not memories:
        return system_prompt  # 无相关记忆，返回原始 prompt

    # 将检索到的记忆追加到 System Prompt 末尾
    examples_section = "\n\n## Relevant Context from Memory\n\n"
    for result in memories:
        examples_section += f"• {result.memory.content}\n"
    return system_prompt + examples_section
```

**RAG 检索流程（两种检索路径）：**

```
路径1：自动检索（每次 LLM 调用前自动执行）
  Agent._send_message()
    → LlmContextEnhancer.enhance_system_prompt()  # 自动检索文本记忆
    → AgentMemory.search_text_memories(query=user_message)
    → 结果拼接到 System Prompt

路径2：工具主动检索（LLM 决定调用工具）
  Agent 工具循环
    → LLM 决定调用 search_saved_correct_tool_uses 工具
    → SearchSavedCorrectToolUsesTool.execute()
    → AgentMemory.search_similar_usage(question=user_question)
    → 返回相似的历史工具使用模式（含 SQL 参数）
```

**层间数据流向：**
- 输入：`user_message`（原始用户问题）、`system_prompt`（原始 System Prompt）
- 输出：增强后的 `system_prompt`（拼接了相关文本记忆）
- 向下层：将增强后的 System Prompt 传入 `LlmRequest`

**核心调度流程（Agent._send_message）：**

```python
# Agent._send_message() - agent.py 第231-1155行（简化版）
async def _send_message(self, request_context, message, conversation_id):
    # Step 1: 用户身份解析
    user = await self.user_resolver.resolve_user(request_context)

    # Step 2: 工作流处理器拦截（/help、/status 等命令）
    if self.workflow_handler:
        workflow_result = await self.workflow_handler.try_handle(self, user, conversation, message)
        if workflow_result.should_skip_llm:
            return  # 命令被工作流处理，跳过 LLM

    # Step 3: 加载/创建会话
    conversation = await self.conversation_store.get_conversation(conversation_id, user)
    conversation.add_message(Message(role="user", content=message))

    # Step 4: 构建 ToolContext
    context = ToolContext(user=user, conversation_id=conversation_id,
                          request_id=request_id, agent_memory=self.agent_memory)

    # Step 5: 获取用户可用工具列表
    tool_schemas = await self.tool_registry.get_schemas(user)

    # Step 6: 构建 System Prompt
    system_prompt = await self.system_prompt_builder.build_system_prompt(user, tool_schemas)

    # Step 7: RAG 增强 System Prompt
    system_prompt = await self.llm_context_enhancer.enhance_system_prompt(
        system_prompt, message, user
    )

    # Step 8: 构建 LLM 请求（含历史消息、工具列表、System Prompt）
    request = await self._build_llm_request(conversation, tool_schemas, user, system_prompt)

    # Step 9: 工具调用循环（最多 max_tool_iterations 次）
    tool_iterations = 0
    while tool_iterations < self.config.max_tool_iterations:
        # 9a: 调用 LLM
        response = await self._send_llm_request(request)  # 或 _handle_streaming_response

        # 9b: 判断是否包含工具调用
        if response.is_tool_call():
            tool_iterations += 1
            # 9c: 逐工具执行
            for tool_call in response.tool_calls:
                result = await self.tool_registry.execute(tool_call, context)
                # 9d: 将工具结果添加到对话历史
                conversation.add_message(Message(role="tool", content=result.result_for_llm))
            # 9e: 重建请求（含工具结果），继续循环
            request = await self._build_llm_request(conversation, tool_schemas, user, system_prompt)
        else:
            # 9f: 无工具调用，LLM 完成最终回复
            yield UiComponent(rich_component=RichTextComponent(content=response.content))
            break
```

```python
# ToolRegistry.execute() - registry.py 第144-278行
async def execute(self, tool_call: ToolCall, context: ToolContext) -> ToolResult:
    # Step 1: 查找工具
    tool = await self.get_tool(tool_call.name)
    if not tool:
        return ToolResult(success=False, error=f"Tool '{tool_call.name}' not found")

    # Step 2: 权限检查（基于 group_memberships 交集）
    if not await self._validate_tool_permissions(tool, context.user):
        # 审计日志：记录拒绝访问
        await self.audit_logger.log_tool_access_check(...)
        return ToolResult(success=False, error="Insufficient group access")

    # Step 3: Pydantic 参数校验
    args_model = tool.get_args_schema()
    validated_args = args_model.model_validate(tool_call.arguments)

    # Step 4: 参数转换/行级安全注入
    transform_result = await self.transform_args(tool, validated_args, context.user, context)
    if isinstance(transform_result, ToolRejection):
        return ToolResult(success=False, error=transform_result.reason)

    # Step 5: 审计日志 - 记录访问成功
    await self.audit_logger.log_tool_access_check(user=context.user, access_granted=True, ...)

    # Step 6: 审计日志 - 记录工具调用
    await self.audit_logger.log_tool_invocation(user=context.user, tool_call=tool_call, ...)

    # Step 7: 执行工具（计时）
    start_time = time.perf_counter()
    result = await tool.execute(context, final_args)
    execution_time_ms = (time.perf_counter() - start_time) * 1000

    # Step 8: 审计日志 - 记录执行结果
    await self.audit_logger.log_tool_result(user=context.user, result=result, ...)
    return result
```

```python
# ToolRegistry._validate_tool_permissions() - registry.py 第98-111行
async def _validate_tool_permissions(self, tool, user):
    tool_access_groups = tool.access_groups
    if not tool_access_groups:
        return True  # 无访问限制，所有人可访问

    user_groups = set(user.group_memberships)
    tool_groups = set(tool_access_groups)
    # 交集非空即允许访问
    return bool(user_groups & tool_groups)
```

```python
# RunSqlTool.execute() - tools/run_sql.py 第56-165行
async def execute(self, context: ToolContext, args: RunSqlToolArgs) -> ToolResult:
    try:
        # Step 1: 通过 SqlRunner 执行 SQL，返回 DataFrame
        df = await self.sql_runner.run_sql(args, context)

        query_type = args.sql.strip().upper().split()[0]

        if query_type == "SELECT":
            # Step 2a: SELECT 查询 - 转换结果
            results_data = df.to_dict("records")
            columns = df.columns.tolist()
            row_count = len(df)

            # 写入 CSV 文件供下游工具使用
            csv_content = df.to_csv(index=False)
            await self.file_system.write_file(filename, csv_content, context)

            # 截断过长结果（超过 1000 字符）
            results_preview = csv_content
            if len(results_preview) > 1000:
                results_preview = results_preview[:1000] + "\n(Results truncated...)"

            # 创建 DataFrameComponent 用于 UI 展示
            ui_component = UiComponent(
                rich_component=DataFrameComponent.from_records(records=results_data, ...),
                simple_component=SimpleTextComponent(text=result),
            )
        else:
            # Step 2b: 非查询语句 - 返回影响行数
            rows_affected = len(df) if not df.empty else 0
            result = f"Query executed successfully. {rows_affected} row(s) affected."

        return ToolResult(success=True, result_for_llm=result,
                          ui_component=ui_component, metadata=metadata)

    except Exception as e:
        # Step 3: 异常处理 - 返回错误信息给 LLM
        error_message = f"Error executing query: {str(e)}"
        return ToolResult(success=False, result_for_llm=error_message,
                          ui_component=NotificationComponent(level="error", ...),
                          error=str(e))
```

```python
# PostgresRunner.run_sql() - integrations/postgres/sql_runner.py 第65-112行
async def run_sql(self, args: RunSqlToolArgs, context: ToolContext) -> pd.DataFrame:
    # 使用连接字符串或参数连接
    if self.connection_string:
        conn = self.psycopg2.connect(self.connection_string)
    else:
        conn = self.psycopg2.connect(**self.connection_params)

    cursor = conn.cursor(cursor_factory=self.psycopg2.extras.RealDictCursor)
    try:
        cursor.execute(args.sql)  # 执行 SQL

        query_type = args.sql.strip().upper().split()[0]
        if query_type == "SELECT":
            rows = cursor.fetchall()
            results_data = [dict(row) for row in rows]
            return pd.DataFrame(results_data)
        else:
            conn.commit()
            rows_affected = cursor.rowcount
            return pd.DataFrame({"rows_affected": [rows_affected]})
    finally:
        cursor.close()
        conn.close()
```
