```python
# 新建文件: src/vanna/integrations/my_llm/llm.py
from vanna.core.llm import LlmService, LlmRequest, LlmResponse, LlmStreamChunk
from vanna.core.tool import ToolCall, ToolSchema
import json
import httpx
from typing import List, Optional, Dict, Any, AsyncGenerator

class MyCustomLlmService(LlmService):
    """自研 LLM 接入示例"""

    def __init__(self, model: str, api_key: str, base_url: str):
        self.model = model
        self.api_key = api_key
        self.base_url = base_url
        self._client = httpx.AsyncClient(
            base_url=base_url,
            headers={"Authorization": f"Bearer {api_key}"},
            timeout=60.0,
        )

    async def send_request(self, request: LlmRequest) -> LlmResponse:
        """非流式请求"""
        payload = self._build_payload(request)

        resp = await self._client.post("/v1/chat/completions", json=payload)
        resp.raise_for_status()
        data = resp.json()

        choice = data["choices"][0]
        message = choice.get("message", {})

        # 提取工具调用
        tool_calls = None
        if "tool_calls" in message:
            tool_calls = [
                ToolCall(
                    id=tc.get("id", ""),
                    name=tc["function"]["name"],
                    arguments=json.loads(tc["function"]["arguments"]),
                )
                for tc in message["tool_calls"]
            ]

        return LlmResponse(
            content=message.get("content"),
            tool_calls=tool_calls,
            finish_reason=choice.get("finish_reason"),
            usage={
                "prompt_tokens": data.get("usage", {}).get("prompt_tokens", 0),
                "completion_tokens": data.get("usage", {}).get("completion_tokens", 0),
                "total_tokens": data.get("usage", {}).get("total_tokens", 0),
            },
        )

    async def stream_request(self, request: LlmRequest) -> AsyncGenerator[LlmStreamChunk, None]:
        """流式请求"""
        payload = self._build_payload(request)
        payload["stream"] = True

        async with self._client.stream("POST", "/v1/chat/completions", json=payload) as resp:
            tc_builders: Dict[int, Dict] = {}
            async for line in resp.aiter_lines():
                if not line.startswith("data: "):
                    continue
                if line == "data: [DONE]":
                    break

                data = json.loads(line[6:])
                delta = data["choices"][0].get("delta", {})

                if delta.get("content"):
                    yield LlmStreamChunk(content=delta["content"])

                # 累积流式 tool_calls
                for tc in delta.get("tool_calls", []) or []:
                    idx = tc.get("index", 0)
                    b = tc_builders.setdefault(idx, {"id": None, "name": None, "arguments": ""})
                    if tc.get("id"): b["id"] = tc["id"]
                    fn = tc.get("function", {})
                    if fn.get("name"): b["name"] = fn["name"]
                    if fn.get("arguments"): b["arguments"] += fn["arguments"]

            # 最终 tool_calls
            final_tc = []
            for b in tc_builders.values():
                if b.get("name"):
                    final_tc.append(ToolCall(
                        id=b.get("id", ""), name=b["name"],
                        arguments=json.loads(b["arguments"] or "{}"),
                    ))
            if final_tc:
                yield LlmStreamChunk(tool_calls=final_tc, finish_reason="stop")

    async def validate_tools(self, tools: List[ToolSchema]) -> List[str]:
        """校验工具 Schema"""
        errors = []
        for t in tools:
            if not t.name or len(t.name) > 64:
                errors.append(f"Invalid tool name: {t.name!r}")
        return errors

    def _build_payload(self, request: LlmRequest) -> Dict[str, Any]:
        """构建 API 请求体（兼容 OpenAI Chat Completions 格式）"""
        messages = []

        if request.system_prompt:
            messages.append({"role": "system", "content": request.system_prompt})

        for m in request.messages:
            msg = {"role": m.role, "content": m.content}
            if m.role == "tool" and m.tool_call_id:
                msg["tool_call_id"] = m.tool_call_id
            elif m.role == "assistant" and m.tool_calls:
                msg["tool_calls"] = [
                    {"id": tc.id, "type": "function",
                     "function": {"name": tc.name, "arguments": json.dumps(tc.arguments)}}
                    for tc in m.tool_calls
                ]
            messages.append(msg)

        payload = {"model": self.model, "messages": messages}

        if request.tools:
            payload["tools"] = [
                {"type": "function", "function": {
                    "name": t.name, "description": t.description, "parameters": t.parameters
                }}
                for t in request.tools
            ]
            payload["tool_choice"] = "auto"

        if request.max_tokens:
            payload["max_tokens"] = request.max_tokens
        if request.temperature is not None:
            payload["temperature"] = request.temperature

        return payload
```

```python
from vanna.capabilities.sql_runner import SqlRunner, RunSqlToolArgs
from vanna.core.tool import ToolContext
import pandas as pd
from typing import Optional

class MyDatabaseRunner(SqlRunner):
    """自定义数据库执行器"""

    def __init__(self, connection_string: Optional[str] = None, **connection_params):
        self.connection_string = connection_string
        self.connection_params = connection_params

    async def run_sql(self, args: RunSqlToolArgs, context: ToolContext) -> pd.DataFrame:
        """执行 SQL 并返回 DataFrame"""
        import my_database_driver  # 替换为实际驱动

        # 建立连接
        if self.connection_string:
            conn = my_database_driver.connect(self.connection_string)
        else:
            conn = my_database_driver.connect(**self.connection_params)

        cursor = conn.cursor()
        try:
            cursor.execute(args.sql)

            query_type = args.sql.strip().upper().split()[0]

            if query_type == "SELECT":
                columns = [desc[0] for desc in cursor.description]
                rows = cursor.fetchall()
                return pd.DataFrame(rows, columns=columns)
            else:
                conn.commit()
                return pd.DataFrame({"rows_affected": [cursor.rowcount]})
        finally:
            cursor.close()
            conn.close()
```

**实现方案1：SQL 注入防护 + 敏感操作拦截**

```python
# 新建文件: src/vanna/integrations/my_security/safe_run_sql.py
from vanna.tools.run_sql import RunSqlTool
from vanna.core.tool import ToolContext, ToolResult
from vanna.capabilities.sql_runner import RunSqlToolArgs
import re

class SafeRunSqlTool(RunSqlTool):
    """带安全规则的 SQL 执行工具"""

    # 禁止的 SQL 关键字
    FORBIDDEN_KEYWORDS = [
        "DROP", "TRUNCATE", "ALTER", "CREATE", "GRANT", "REVOKE",
        "EXEC", "EXECUTE", "INTO OUTFILE", "INTO DUMPFILE",
    ]

    # 禁止的 SQL 模式（正则）
    FORBIDDEN_PATTERNS = [
        r"--\s*$",                    # 行尾注释（可能用于绕过检测）
        r"/\*.*\*/",                  # 块注释
        r"UNION\s+SELECT",            # UNION 注入
        r"OR\s+1\s*=\s*1",           # 条件注入
        r"'\s*OR\s+'1'\s*=\s*'1",   # 字符串注入
    ]

    async def execute(self, context: ToolContext, args: RunSqlToolArgs) -> ToolResult:
        """执行前先做安全检查"""
        sql = args.sql

        # 1. 检查禁止关键字
        sql_upper = sql.upper().strip()
        for kw in self.FORBIDDEN_KEYWORDS:
            if sql_upper.startswith(kw) or f" {kw} " in sql_upper:
                return ToolResult(
                    success=False,
                    result_for_llm=f"Security: Operation '{kw}' is not allowed. "
                                   f"Only SELECT and DML statements are permitted.",
                    error=f"Forbidden keyword: {kw}",
                )

        # 2. 检查 SQL 注入模式
        for pattern in self.FORBIDDEN_PATTERNS:
            if re.search(pattern, sql, re.IGNORECASE):
                return ToolResult(
                    success=False,
                    result_for_llm=f"Security: Potentially dangerous SQL pattern detected.",
                    error=f"Suspicious pattern: {pattern}",
                )

        # 3. 仅允许 SELECT / INSERT / UPDATE / DELETE
        allowed_prefixes = ["SELECT", "INSERT", "UPDATE", "DELETE", "WITH", "EXPLAIN"]
        if not any(sql_upper.startswith(p) for p in allowed_prefixes):
            return ToolResult(
                success=False,
                result_for_llm=f"Only SELECT, INSERT, UPDATE, DELETE queries are allowed.",
                error="Query type not allowed",
            )

        # 4. 通过安全检查，执行原始逻辑
        return await super().execute(context, args)
```

**实现方案2：行级安全（RLS）注入**

```python
# 在 ToolRegistry.transform_args() 中注入行级安全逻辑
# 参考: core/registry.py 第220-250行

class RlsRunSqlTool(RunSqlTool):
    """带行级安全的 SQL 执行工具"""

    async def transform_args(self, args: RunSqlToolArgs, user, context) -> RunSqlToolArgs:
        """在 SQL 中注入行级安全过滤条件"""
        sql = args.sql

        # 获取用户所属的租户 ID
        tenant_id = user.metadata.get("tenant_id", "")
        if not tenant_id:
            return args  # 无租户信息，不注入

        # 对 SELECT 语句注入 WHERE tenant_id = ?
        sql_upper = sql.upper().strip()
        if sql_upper.startswith("SELECT"):
            if "WHERE" in sql_upper:
                # 已有 WHERE 子句，追加 AND
                sql = sql.replace("WHERE", f"WHERE tenant_id = '{tenant_id}' AND ", 1)
            else:
                # 无 WHERE 子句，在末尾添加
                # 注意：需要处理 GROUP BY / ORDER BY / LIMIT 等
                insertion_point = self._find_where_insertion_point(sql)
                sql = sql[:insertion_point] + f" WHERE tenant_id = '{tenant_id}'" + sql[insertion_point:]

        return RunSqlToolArgs(sql=sql)

    def _find_where_insertion_point(self, sql: str) -> int:
        """找到 WHERE 子句的插入位置"""
        for keyword in ["GROUP BY", "ORDER BY", "LIMIT", "OFFSET", "HAVING"]:
            idx = sql.upper().find(keyword)
            if idx != -1:
                return idx
        return len(sql)  # 末尾
```

```javascript
const reader = response.body.getReader();
const decoder = new TextDecoder();

while (true) {
    const { done, value } = await reader.read();
    if (done) break;

    const text = decoder.decode(value);
    const lines = text.split('\n').filter(l => l.startsWith('data: '));

    for (const line of lines) {
        const chunk = JSON.parse(line.slice(6));
        // 根据 chunk.type 渲染自定义 UI
        switch (chunk.type) {
            case 'rich_text':
                renderText(chunk.component);
                break;
            case 'dataframe':
                renderTable(chunk.component);
                break;
            case 'status_bar_update':
                updateStatus(chunk.component);
                break;
            case 'done':
                finishResponse();
                break;
        }
    }
}
```

```python
# 新建文件: src/vanna/integrations/my_auth/resolver.py
from vanna.core.user.resolver import UserResolver
from vanna.core.user.request_context import RequestContext
from vanna.core.user.models import User
import jwt

class JwtUserResolver(UserResolver):
    """基于 JWT Token 的用户解析器"""

    def __init__(self, secret_key: str, algorithm: str = "HS256"):
        self.secret_key = secret_key
        self.algorithm = algorithm

    async def resolve_user(self, request_context: RequestContext) -> User:
        # 从 Authorization Header 中提取 Token
        auth_header = request_context.get_header("Authorization")
        if not auth_header or not auth_header.startswith("Bearer "):
            raise PermissionError("Missing or invalid Authorization header")

        token = auth_header[7:]  # 去掉 "Bearer " 前缀

        try:
            payload = jwt.decode(token, self.secret_key, algorithms=[self.algorithm])
        except jwt.ExpiredSignatureError:
            raise PermissionError("Token expired")
        except jwt.InvalidTokenError:
            raise PermissionError("Invalid token")

        return User(
            id=payload["sub"],
            email=payload.get("email"),
            name=payload.get("name"),
            group_memberships=payload.get("groups", []),
            metadata={
                "tenant_id": payload.get("tenant_id", ""),
                "role": payload.get("role", "user"),
            },
        )
```

```python
from vanna.tools.run_sql import RunSqlTool

# 管理员工具：可执行所有 SQL
admin_sql_tool = RunSqlTool(sql_runner=db_runner)
admin_sql_tool.access_groups = ["admin"]  # 仅 admin 组可访问

# 普通用户工具：仅可执行 SELECT
class ReadOnlySqlTool(RunSqlTool):
    access_groups = ["user", "analyst"]  # 普通用户可访问

    async def execute(self, context, args):
        sql_upper = args.sql.strip().upper()
        if not sql_upper.startswith("SELECT") and not sql_upper.startswith("WITH"):
            return ToolResult(
                success=False,
                result_for_llm="You only have read-only access. Only SELECT queries are allowed.",
            )
        return await super().execute(context, args)

registry.register(admin_sql_tool)
registry.register(ReadOnlySqlTool(sql_runner=db_runner))
```

```python
from vanna.core.audit.base import AuditLogger

class DatabaseAuditLogger(AuditLogger):
    """将审计日志写入数据库"""

    def __init__(self, db_connection):
        self.db = db_connection

    async def log_tool_access_check(self, user, tool_name, access_granted, reason=""):

            "INSERT INTO audit_log (event_type, user_id, tool_name, success, reason, timestamp) "
            "VALUES ('tool_access_check', ?, ?, ?, ?, NOW())",
            [user.id, tool_name, access_granted, reason],
        )

    async def log_tool_invocation(self, user, tool_call, context):
        await self.db.execute(
            "INSERT INTO audit_log (event_type, user_id, tool_name, arguments, timestamp) "
            "VALUES ('tool_invocation', ?, ?, ?, NOW())",
            [user.id, tool_call.name, json.dumps(tool_call.arguments)],
        )

    async def log_tool_result(self, user, result, execution_time_ms):
        await self.db.execute(
            "INSERT INTO audit_log (event_type, user_id, success, execution_time_ms, timestamp) "
            "VALUES ('tool_result', ?, ?, ?, NOW())",
            [user.id, result.success, execution_time_ms],
        )

    async def log_ui_feature_access(self, user, feature_name, allowed):
        await self.db.execute(
            "INSERT INTO audit_log (event_type, user_id, feature_name, success, timestamp) "
```
